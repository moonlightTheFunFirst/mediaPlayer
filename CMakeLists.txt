cmake_minimum_required(VERSION 3.21)
project(mediaPlayer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets)

add_library(playback STATIC src/MediaBackend.cpp src/MediaBackend.h src/ThumbnailProvider.cpp src/ThumbnailProvider.h)
target_link_libraries(playback PUBLIC Qt6::Multimedia)
set(PLAYER_UI_SOURCES src/MainWindow.cpp src/MainWindow.h src/SeekPreview.cpp src/SeekPreview.h assets/orange.qrc)
qt_add_executable(mediaPlayer WIN32 src/main.cpp ${PLAYER_UI_SOURCES})
set_target_properties(mediaPlayer PROPERTIES OUTPUT_NAME Orange)
if(WIN32)
    target_sources(mediaPlayer PRIVATE windows/resources/orange.rc)
    target_include_directories(mediaPlayer PRIVATE windows/resources)
endif()
target_link_libraries(mediaPlayer PRIVATE playback Qt6::Widgets Qt6::MultimediaWidgets)
if(MSVC)
    target_compile_options(playback PRIVATE /utf-8 /W4)
    target_compile_options(mediaPlayer PRIVATE /utf-8 /W4)
endif()

option(MEDIAPLAYER_BUILD_TESTS "Build playback integration checks" OFF)
if(MEDIAPLAYER_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    enable_testing()
    qt_add_executable(playbackTests tests/PlaybackTests.cpp ${PLAYER_UI_SOURCES})
    target_link_libraries(playbackTests PRIVATE playback Qt6::Test Qt6::Widgets Qt6::MultimediaWidgets)
    if(MSVC)
        target_compile_options(playbackTests PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME playbackTests COMMAND playbackTests)
endif()
